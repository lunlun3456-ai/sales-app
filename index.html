// ========================================== // CONFIGURATION: Set your Sheet Names here // ========================================== const SALES_SHEET_NAME = "Sales"; const USERS_SHEET_NAME = "Users"; function doGet(e) { const action = e.parameter.action; const ss = SpreadsheetApp.getActiveSpreadsheet(); // 1. FETCH SALES RECORDS if (action === "getSales") { const sheet = ss.getSheetByName(SALES_SHEET_NAME); if (!sheet) return createJsonResponse({ error: "Sales sheet not found" }); const data = sheet.getDataRange().getValues(); const salesArray = []; for (let i = 1; i < data.length; i++) { if (!data[i][0]) continue; // Skip empty rows salesArray.push({ id: data[i][0], store: data[i][1], cash: Number(data[i][2]) || 0, qr: Number(data[i][3]) || 0, card: Number(data[i][4]) || 0, bank: Number(data[i][5]) || 0, total: Number(data[i][6]) || 0, remark: data[i][7], submitter: data[i][8], role: data[i][9], datetime: data[i][10], dateKey: data[i][11], month: data[i][12] }); } return createJsonResponse({ sales: salesArray }); } // 2. FETCH USER PROFILE DATABASE if (action === "getUsers") { const sheet = ss.getSheetByName(USERS_SHEET_NAME); if (!sheet) return createJsonResponse({ error: "Users sheet not found" }); const data = sheet.getDataRange().getValues(); const usersObj = {}; for (let i = 1; i < data.length; i++) { const uname = String(data[i][0]).trim().toLowerCase(); if (!uname) continue; usersObj[uname] = { name: data[i][1], pin: String(data[i][2]), store: data[i][3], role: data[i][4], status: data[i][5] }; } return createJsonResponse(usersObj); } return createJsonResponse({ error: "Invalid action" }); } function doPost(e) { let rawData; try { rawData = JSON.parse(e.postData.contents); } catch (err) { return createJsonResponse({ error: "Malformed JSON payload" }); } const action = rawData.action; const ss = SpreadsheetApp.getActiveSpreadsheet(); // A. SUBMIT OR OVERWRITE DAILY SALES REPORT if (action === "submitSales") { const sheet = ss.getSheetByName(SALES_SHEET_NAME); const data = sheet.getDataRange().getValues(); let targetRow = -1; // Check if matching dateKey + store already exists to overwrite it for (let i = 1; i < data.length; i++) { if (data[i][11] === rawData.dateKey && data[i][1] === rawData.store) { targetRow = i + 1; break; } } const rowValues = [ rawData.id, rawData.store, rawData.cash, rawData.qr, rawData.card, rawData.bank, rawData.total, rawData.remark, rawData.submitter, rawData.role, rawData.datetime, rawData.dateKey, rawData.month ]; if (targetRow !== -1) { sheet.getRange(targetRow, 1, 1, rowValues.length).setValues([rowValues]); } else { sheet.appendRow(rowValues); } return createJsonResponse({ success: true, msg: "Sales logged successfully" }); } // B. REGISTRATION REQUEST if (action === "registerRequest") { const sheet = ss.getSheetByName(USERS_SHEET_NAME); const uname = String(rawData.username).trim().toLowerCase(); // Prevent duplicated username entries const data = sheet.getDataRange().getValues(); for (let i = 1; i < data.length; i++) { if (String(data[i][0]).toLowerCase() === uname) { return createJsonResponse({ error: "Username already exists" }); } } sheet.appendRow([uname, rawData.name, String(rawData.pin), rawData.store, rawData.role, "pending"]); return createJsonResponse({ success: true }); } // C. APPROVE ACCOUNT USER if (action === "approveUser") { const sheet = ss.getSheetByName(USERS_SHEET_NAME); const data = sheet.getDataRange().getValues(); const uname = String(rawData.username).trim().toLowerCase(); for (let i = 1; i < data.length; i++) { if (String(data[i][0]).toLowerCase() === uname) { sheet.getRange(i + 1, 6).setValue("active"); return createJsonResponse({ success: true }); } } return createJsonResponse({ error: "User not found" }); } // D. REJECT / REMOVE PENDING ENTRY if (action === "rejectUser" || action === "deleteUser") { const sheet = ss.getSheetByName(USERS_SHEET_NAME); const data = sheet.getDataRange().getValues(); const uname = String(rawData.username).trim().toLowerCase(); for (let i = 1; i < data.length; i++) { if (String(data[i][0]).toLowerCase() === uname) { sheet.deleteRow(i + 1); return createJsonResponse({ success: true }); } } return createJsonResponse({ error: "User not found" }); } // E. MANUAL SAVE / UPDATE USER DATA FROM ADMIN MODAL if (action === "saveUser") { const sheet = ss.getSheetByName(USERS_SHEET_NAME); const data = sheet.getDataRange().getValues(); const origKey = String(rawData.originalKey).trim().toLowerCase(); const uname = String(rawData.username).trim().toLowerCase(); let targetRow = -1; for (let i = 1; i < data.length; i++) { if (String(data[i][0]).toLowerCase() === (origKey || uname)) { targetRow = i + 1; break; } } const values = [uname, rawData.name, String(rawData.pin), rawData.store, rawData.role, rawData.status]; if (targetRow !== -1) { sheet.getRange(targetRow, 1, 1, values.length).setValues([values]); } else { sheet.appendRow(values); } return createJsonResponse({ success: true }); } return createJsonResponse({ error: "Invalid action type" }); } function createJsonResponse(obj) { return ContentService.createTextOutput(JSON.stringify(obj)) .setMimeType(ContentService.MimeType.JSON); }
